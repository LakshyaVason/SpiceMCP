# Standalone deps for the sibling benchmark/utility scripts.
#
# These are intentionally kept out of the app's dependency tree: the token-usage
# comparison runs externally, so these scripts only need a Bedrock client.
#
#   python -m venv scripts/.venv
#   scripts\.venv\Scripts\activate
#   pip install -r scripts/requirements.txt
#
# (The project .venv at the repo root already satisfies these, so activating that
# instead works fine for one-off runs.)

# probe_tool_calling.py talks to Bedrock through the Anthropic SDK; the [bedrock] extra
# brings boto3/botocore, which is also what list_bedrock_models.py uses directly.
anthropic[bedrock]>=1.3
