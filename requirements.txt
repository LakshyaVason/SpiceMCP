# SPICE MCP client - runtime dependencies

# MCP server + stdio client. NOTE: v2.x is a rewrite of the v1 API.
# v1 used FastMCP; v2 uses mcp.server.MCPServer. Do not downgrade below 2.1
# without rewriting spice_mcp_server/server.py.
mcp>=2.1,<3

# LTspice .asc/.netlist parsing, .raw waveform reading, .log parsing.
spicelib>=1.6.3,<2

# Desktop window for the app (uses the OS webview; no Node/Electron toolchain).
pywebview>=6.2

# TAMU AI Chat is OpenAI-compatible and called over plain HTTP.
requests>=2.32

# Loads TAMU_API_KEY from a git-ignored .env.
python-dotenv>=1.0

# Process handling: killing a hung LTspice batch run, and detecting a running LTspice GUI
# so an applied patch can warn that the GUI will overwrite it. It has always arrived as a
# spicelib transitive; declared here because shipped features now depend on it directly.
psutil>=7

# Tests.
pytest>=8
